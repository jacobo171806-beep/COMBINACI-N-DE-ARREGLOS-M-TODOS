void CrearVector(string pais, int pruebas) {
    int* tipoMedalla = new int[pruebas];
    LlenarVector(tipoMedalla, pruebas, pais);
}

void LlenarVector(int tipo[], int pruebas, string pais) {
    srand(time(NULL));
    for (int i = 0; i < pruebas; i++) {
        tipo[i] = rand() % 4;
    }
    RealizarConteo(tipo, pais, pruebas);
}

void RealizarConteo(int tipo[], string pais, int pruebas) {
    int oros = 0, platas = 0, bronces = 0;
    for (int i = 0; i < pruebas; i++) {
        switch (tipo[i]) {
            case 1: oros++; break;
            case 2: platas++; break;
            case 3: bronces++; break;
        }
    }
    ImprimirReporte(oros, platas, bronces, pais, pruebas);
}

void ImprimirReporte(int oro, int pl, int br, string pais, int pruebas) {
    cout << "\n================================" << endl;
    cout << "Juegos olimpicos de Tokio" << endl;
    cout << "Year: 2020" << endl;
    cout << "Pais: " << pais << endl;
    cout << "Resultado Total Medallero:" << endl;
    cout << "Oros Platas Bronces Total" << endl;
    cout << " " << oro << " " << pl << " " << br << " " << (oro + pl + br) << endl;
    cout << "Total Pruebas: " << pruebas << endl;
    cout << "================================" << endl;
}
